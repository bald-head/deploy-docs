## 概述

`ui-pc` 是 **High 优先级** Skill，负责按已发现的组件库、路由、状态与设计令牌实现 PC/管理端界面。

**适用场景**：
- PC/Web 管理后台页面
- 桌面端应用界面
- 数据看板和报表
- 管理系统界面

**不适用场景**：
- 移动端/跨端应用 → 使用 `ui-app`
- 设计基线制定 → 使用 `frontend-design`
- 已有页面精修 → 使用 `frontend-polish`
- 后端 API → 使用 `api-development`

## 快速开始

```vue
<template>
  <div class="user-list">
    <!-- 搜索区 -->
    <el-card class="search-card">
      <el-form :model="searchForm" inline>
        <el-form-item label="邮箱">
          <el-input v-model="searchForm.email" placeholder="请输入邮箱" clearable />
        </el-form-item>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import type { FormInstance } from 'element-plus'
import { userApi } from '@/api/user'
import type { User, UserSearchParams } from '@/types/user'

// 搜索表单
const searchForm = reactive<UserSearchParams>({
  email: '',
  phone: '',
  status: ''
})

// 用户列表
const loading = ref(false)
const userList = ref<User[]>([])

// 分页
const pagination = reactive({
  page: 1,
  pageSize: 10,
  total: 0
})

// 编辑对话框
const editDialogVisible = ref(false)
const editFormRef = ref<FormInstance>()
const editForm = reactive<Partial<User>>({
  id: '',
  name: '',
  email: '',
  phone: '',
  status: 'ACTIVE'
})
const submitting = ref(false)

// 表单验证规则
const editRules = {
  name: [
    { required: true, message: '请输入姓名', trigger: 'blur' },
    { min: 2, max: 50, message: '长度在 2 到 50 个字符', trigger: 'blur' }
  ],
  email: [
    { required: true, message: '请输入邮箱', trigger: 'blur' },
    { type: 'email', message: '请输入正确的邮箱格式', trigger: 'blur' }
  ],
  phone: [
    { required: true, message: '请输入手机号', trigger: 'blur' },
    { pattern: /^1[3-9]\d{9}$/, message: '请输入正确的手机号', trigger: 'blur' }
  ],
  status: [
    { required: true, message: '请选择状态', trigger: 'change' }
  ]
}

// 获取用户列表
const loadUserList = async () => {
  try {
    loading.value = true
    const res = await userApi.list({
      ...searchForm,
      page: pagination.page,
      pageSize: pagination.pageSize
    })
    userList.value = res.data.records
    pagination.total = res.data.total
  } catch (error) {
    ElMessage.error('加载用户列表失败')
  } finally {
    loading.value = false
  }
}

// 搜索
const handleSearch = () => {
  pagination.page = 1
  loadUserList()
}

// 重置
const handleReset = () => {
  searchForm.email = ''
  searchForm.phone = ''
  searchForm.status = ''
  handleSearch()
}

// 分页
const handleSizeChange = (size: number) => {
  pagination.pageSize = size
  loadUserList()
}

const handlePageChange = (page: number) => {
  pagination.page = page
  loadUserList()
}

// 编辑
const handleEdit = (row: User) => {
  Object.assign(editForm, row)
  editDialogVisible.value = true
}

// 提交编辑
const handleSubmit = async () => {
  if (!editFormRef.value) return
  
  await editFormRef.value.validate()
  
  try {
    submitting.value = true
    await userApi.update(editForm.id!, editForm)
    ElMessage.success('修改成功')
    editDialogVisible.value = false
    loadUserList()
  } catch (error) {
    ElMessage.error('修改失败')
  } finally {
    submitting.value = false
  }
}

// 删除
const handleDelete = async (row: User) => {
  try {
    await ElMessageBox.confirm(
      `确定删除用户 ${row.name} 吗？`,
      '提示',
      {
        type: 'warning'
      }
    )
    
    await userApi.delete(row.id)
    ElMessage.success('删除成功')
    loadUserList()
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('删除失败')
    }
  }
}

// 状态相关
const getStatusType = (status: string) => {
  const map: Record<string, any> = {
    ACTIVE: 'success',
    INACTIVE: 'warning',
    BANNED: 'danger'
  }
  return map[status] || 'info'
}

const getStatusLabel = (status: string) => {
  const map: Record<string, string> = {
    ACTIVE: '激活',
    INACTIVE: '停用',
    BANNED: '禁用'
  }
  return map[status] || status
}

// 初始化
onMounted(() => {
  loadUserList()
})
</script>
```

3. **定义 API 契约**：
```typescript
// src/api/user.ts
import request from '@/utils/request'
import type { User, UserSearchParams, PageResult } from '@/types/user'

export const userApi = {
  // 查询用户列表
  list(params: UserSearchParams) {
    return request.get<PageResult<User>>('/api/v1/users', { params })
  },

  // 更新用户
  update(id: string, data: Partial<User>) {
    return request.put<void>(`/api/v1/users/${id}`, data)
  },

  // 删除用户
  delete(id: string) {
    return request.delete<void>(`/api/v1/users/${id}`)
  }
}
```

4. **定义类型**：
```typescript
// src/types/user.ts
export interface User {

  email?: string
  phone?: string
  status?: string
  page?: number
  pageSize?: number
}

export interface PageResult<T> {
  records: T[]
  total: number
  page: number
  pageSize: number
}
```

```vue
import { ref, reactive, onMounted, onUnmounted } from 'vue'
import * as echarts from 'echarts'
import type { ECharts } from 'echarts'
import { dashboardApi } from '@/api/dashboard'
import type { SalesMetrics, TrendData, TopProduct } from '@/types/dashboard'

// 关键指标
const metrics = reactive<SalesMetrics>({
  todaySales: 0,
  orderCount: 0,
  avgOrderValue: 0,
  salesGrowth: 0,
  orderGrowth: 0,
  avgGrowth: 0,
  totalGrowth: 0

const mapChartRef = ref<HTMLElement>()

let mapChart: ECharts | null = null

// 加载数据
const loadData = async () => {
  try {
    const [startDate, endDate] = dateRange.value
    const res = await dashboardApi.getSalesData({
      startDate: startDate.toISOString().split('T')[0],
      endDate: endDate.toISOString().split('T')[0]

      type: 'category',
      data: data.map(d => d.date)
    },
    yAxis: [
      {
        type: 'value',
        name: '销售额（元）',
        position: 'left'
      },
      {
        type: 'value',
        name: '订单量',
        position: 'right'
      }
```
