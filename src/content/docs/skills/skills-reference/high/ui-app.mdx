## 概述

`ui-app` 是 **High 优先级** Skill，负责按已发现的平台、框架、导航与视觉令牌实现跨端应用界面。

**适用场景**：
- 移动端 App 页面
- 跨端应用（React Native/Flutter/UniApp/Taro）
- 小程序页面
- H5 移动页面

**不适用场景**：
- PC/管理端界面 → 使用 `ui-pc`
- 设计基线制定 → 使用 `frontend-design`
- 已有页面精修 → 使用 `frontend-polish`
- 后端 API → 使用 `api-development`

## 快速开始

### 实现移动端页面的完整流程

```bash
# 1. 确认设计基线已完成
# 查看 TODO 中的页面结构、交互、状态定义

# 2. 发现项目结构
# 查找平台、框架、导航、组件库

# 3. 开始实现
"实现用户登录页面"
"实现商品列表页"

# 4. 接入真实 API
# 确认后端接口已就绪，接入真实数据

# 5. 验证
npm run dev:h5          # H5 预览
npm run dev:mp-weixin   # 微信小程序
npm run dev:app         # App 预览

# 6. Review
/code-review
```

```yaml

功能: 手机号登录、验证码登录

状态: default, loading, error, success
```

**实现步骤**：

1. **发现项目结构**：
```bash
# 查找项目配置
rg --files | rg 'manifest.json|pages.json'

# 查找组件和页面
rg --files | rg 'pages/|components/'

# 查找相邻页面
rg -l '<template>' src/pages/
```

2. **创建页面文件**：
```vue
<!-- src/pages/login/index.vue -->
<template>
  <view class="login-page">
    <!-- 顶部装饰 -->
    <view class="header">
      <image class="logo" src="/static/logo.png" mode="aspectFit" />
      <text class="title">欢迎登录</text>
    </view>

    <!-- 登录表单 -->
    <view class="form">
      <!-- 手机号输入 -->
      <view class="form-item">
        <view class="input-wrapper">
          <input
            v-model="form.phone"
            type="number"
            maxlength="11"

            placeholder-class="placeholder"
            class="input"
            @input="handlePhoneInput"
          />

            placeholder-class="placeholder"
            class="input"

            @click="handleSendCode"

<script setup lang="ts">
import { ref, computed } from 'vue'
import { authApi } from '@/api/auth'
import { useUserStore } from '@/stores/user'

const userStore = useUserStore()

// 表单数据
const form = ref({
  phone: '',
  code: ''
})

// 错误信息
const errors = ref({
  phone: '',
  code: ''
})

// 状态
const loading = ref(false)
const countdown = ref(0)
const agreed = ref(false)
let timer: number | null = null

// 验证手机号
const validatePhone = (phone: string) => {
  if (!phone) {
    return '请输入手机号'
  }
  if (!/^1[3-9]\d{9}$/.test(phone)) {
    return '请输入正确的手机号'
  }
  return ''
}

// 验证验证码
const validateCode = (code: string) => {
  if (!code) {
    return '请输入验证码'
  }
  if (code.length !== 6) {
    return '验证码为6位数字'
  }
  return ''
}

// 能否发送验证码
const canSendCode = computed(() => {
  return form.value.phone.length === 11 && !validatePhone(form.value.phone)
})

// 能否提交
const canSubmit = computed(() => {
  return (
    form.value.phone.length === 11 &&
    form.value.code.length === 6 &&
    agreed.value &&
    !loading.value
  )
})

// 验证码按钮文本
const codeButtonText = computed(() => {
  if (countdown.value > 0) {
    return `${countdown.value}秒后重试`
  }
  return '获取验证码'
})

// 手机号输入
const handlePhoneInput = () => {
  errors.value.phone = ''
}

// 清空手机号
const clearPhone = () => {
  form.value.phone = ''
  errors.value.phone = ''
}

// 发送验证码
const handleSendCode = async () => {
  const error = validatePhone(form.value.phone)
  if (error) {
    errors.value.phone = error
    return
  }

  try {
    await authApi.sendCode({ phone: form.value.phone })
    uni.showToast({
      title: '验证码已发送',
      icon: 'success'
    })

    // 开始倒计时
    countdown.value = 60
    timer = setInterval(() => {
      countdown.value--
      if (countdown.value <= 0) {
        clearInterval(timer!)
        timer = null
      }
    }, 1000) as any
  } catch (error: any) {
    uni.showToast({
      title: error.message || '发送失败',
      icon: 'none'
    })
  }
}

// 登录
const handleLogin = async () => {
  // 验证
  errors.value.phone = validatePhone(form.value.phone)
  errors.value.code = validateCode(form.value.code)

  if (errors.value.phone || errors.value.code) {
    return
  }

  if (!agreed.value) {
    uni.showToast({
      title: '请先同意用户协议和隐私政策',
      icon: 'none'
    })
    return
  }

  try {
    loading.value = true
    const res = await authApi.login({
      phone: form.value.phone,
      code: form.value.code
    })

    // 保存用户信息和 Token
    await userStore.setToken(res.data.token)
    await userStore.fetchUserInfo()

    uni.showToast({
      title: '登录成功',
      icon: 'success'
    })

    // 延迟跳转
    setTimeout(() => {
      uni.switchTab({
        url: '/pages/index/index'
      })
    }, 500)
  } catch (error: any) {
    uni.showToast({
      title: error.message || '登录失败',
      icon: 'none'
    })
  } finally {
    loading.value = false
  }
}

// 协议勾选
const handleAgreementChange = (e: any) => {
  agreed.value = e.detail.value.length > 0
}

// 跳转协议页
const goToAgreement = (type: 'user' | 'privacy') => {
  uni.navigateTo({
    url: `/pages/agreement/index?type=${type}`
  })
}

// 微信登录
const handleWechatLogin = () => {
  // #ifdef MP-WEIXIN
  uni.getUserProfile({
    desc: '用于完善会员资料',
    success: async (res) => {
      try {
        const loginRes = await uni.login()
        const authRes = await authApi.wechatLogin({
          code: loginRes.code,
          userInfo: res.userInfo
        })
        
        await userStore.setToken(authRes.data.token)
        await userStore.fetchUserInfo()
        
        uni.switchTab({
          url: '/pages/index/index'
        })
      } catch (error: any) {
        uni.showToast({
          title: error.message || '登录失败',
          icon: 'none'
        })
      }
    }
  })
  // #endif

  // #ifdef APP-PLUS
  uni.getProvider({
    service: 'oauth',
    success: (res) => {
      if (res.provider.includes('weixin')) {
        uni.login({
          provider: 'weixin',
          success: async (loginRes) => {
            try {
              const authRes = await authApi.wechatLogin({
                code: loginRes.code
              })
              
              await userStore.setToken(authRes.data.token)
              await userStore.fetchUserInfo()
              
              uni.switchTab({
                url: '/pages/index/index'
              })
            } catch (error: any) {
              uni.showToast({
                title: error.message || '登录失败',
                icon: 'none'
              })
            }
          }
        })
      }
    }
  })
  // #endif
}

// Apple 登录
const handleAppleLogin = () => {
  // #ifdef APP-PLUS
  uni.getProvider({
    service: 'oauth',
    success: (res) => {
      if (res.provider.includes('apple')) {
        uni.login({
          provider: 'apple',
          success: async (loginRes) => {
            try {
              const authRes = await authApi.appleLogin({
                identityToken: loginRes.identityToken
              })
              
              await userStore.setToken(authRes.data.token)
              await userStore.fetchUserInfo()
              
              uni.switchTab({
                url: '/pages/index/index'
              })
            } catch (error: any) {
              uni.showToast({
                title: error.message || '登录失败',
                icon: 'none'
              })
            }
          }
        })
      }
    }
  })
  // #endif
}

// 页面卸载
onUnmounted(() => {
  if (timer) {
    clearInterval(timer)
    timer = null
  }
})
</script>
```

**实现步骤**：

```vue
<!-- src/pages/goods/list.vue -->
<template>
  <view class="goods-list-page">
    <!-- 搜索栏 -->
    <view class="search-bar">
      <view class="search-input" @click="goToSearch">
        <uni-icons type="search" size="18" color="#9ca3af" />

<script setup lang="ts">
import { ref, reactive } from 'vue'
import { goodsApi } from '@/api/goods'
import type { Goods, Category } from '@/types/goods'

// 分类列表
const categories = ref<Category[]>([
  { id: '', name: '全部' },
  { id: '1', name: '手机' },
  { id: '2', name: '电脑' },
  { id: '3', name: '数码' }
])
const currentCategory = ref('')

// 商品列表
const goodsList = ref<Goods[]>([])
const loading = ref(false)
const hasMore = ref(true)
const loadMoreStatus = ref<'more' | 'loading' | 'noMore'>('more')

// 分页
const pagination = reactive({
  page: 1,
  pageSize: 20
})

// 筛选
const showFilter = ref(false)
const filter = reactive({
  minPrice: '',
  maxPrice: '',
  sortBy: 'default'
})

const sortOptions = [
  { label: '默认', value: 'default' },
  { label: '价格升序', value: 'price_asc' },
  { label: '价格降序', value: 'price_desc' },
  { label: '销量', value: 'sales' }
]

// 加载商品列表
const loadGoodsList = async (reset = false) => {
  if (loading.value) return

  if (reset) {
    pagination.page = 1
    goodsList.value = []
    hasMore.value = true
  }

  try {
    loading.value = true
    loadMoreStatus.value = 'loading'

    const res = await goodsApi.list({
      categoryId: currentCategory.value,
      minPrice: filter.minPrice ? Number(filter.minPrice) : undefined,
      maxPrice: filter.maxPrice ? Number(filter.maxPrice) : undefined,
      sortBy: filter.sortBy,
      page: pagination.page,
      pageSize: pagination.pageSize
    })

    if (reset) {
      goodsList.value = res.data.records
    } else {
      goodsList.value.push(...res.data.records)
    }

    hasMore.value = goodsList.value.length < res.data.total
    loadMoreStatus.value = hasMore.value ? 'more' : 'noMore'
  } catch (error: any) {
    uni.showToast({
      title: error.message || '加载失败',
      icon: 'none'
    })
    loadMoreStatus.value = 'more'
  } finally {
    loading.value = false
  }
}

// 分类切换
const handleCategoryChange = (categoryId: string) => {
  currentCategory.value = categoryId
  loadGoodsList(true)
}

// 加载更多
const handleLoadMore = () => {
  if (!hasMore.value || loading.value) return
  pagination.page++
  loadGoodsList()
}

// 重置筛选
const handleResetFilter = () => {
  filter.minPrice = ''
  filter.maxPrice = ''
  filter.sortBy = 'default'
}

// 确认筛选
const handleConfirmFilter = () => {
  showFilter.value = false
  loadGoodsList(true)
}

// 跳转搜索
```

## 关键规则

### 1. 平台条件编译

```vue
<!-- 正确: 使用条件编译 -->
<!-- #ifdef MP-WEIXIN -->
<button open-type="getUserProfile">微信授权</button>
<!-- #endif -->

<!-- #ifdef APP-PLUS -->
<button @click="handleAppLogin">App 登录</button>
<!-- #endif -->

<!-- 错误: 不区分平台 -->
<button @click="login">登录</button>
```

### 2. 安全区域适配

```vue
<!-- 正确: 考虑安全区域 -->
<view class="page" :style="{ paddingTop: statusBarHeight + 'px' }">
  <view class="content" :style="{ paddingBottom: safeAreaBottom + 'px' }">
    <!-- 内容 -->
  </view>
</view>

<!-- 错误: 不考虑安全区域 -->
<view class="page">
  <!-- 内容可能被刘海屏遮挡 -->
</view>
```

### 3. 单位使用 rpx

```scss
// 正确: 使用 rpx
.button {
  width: 686rpx;      // 自动适配不同屏幕
  height: 88rpx;
  font-size: 32rpx;
}

// 错误: 使用固定 px
.button {
  width: 343px;       // 不同屏幕显示不一致
  height: 44px;
  font-size: 16px;
}
```

### 4. 图片优化

```vue
<!-- 正确: 使用 mode 属性 -->
<image
  class="avatar"
  :src="avatar"
  mode="aspectFill"
  lazy-load
/>

<!-- 错误: 不指定 mode -->
<image
  class="avatar"
  :src="avatar"
/>
```

### 4. 请求取消

```typescript
// 正确: 可取消请求
let abortController: AbortController | null = null

const loadData = async () => {
  // 取消上次请求
  if (abortController) {
    abortController.abort()
  }
  
  abortController = new AbortController()
  
  try {
    const res = await api.getData({ signal: abortController.signal })
    // 处理数据
  } catch (error) {
    if (error.name !== 'AbortError') {
      // 处理错误
    }
  }
}

// 错误: 不取消请求
const loadData = async () => {
  const res = await api.getData()
  // 可能被旧请求覆盖
}
```
