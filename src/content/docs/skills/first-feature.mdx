### 2.2 实现 API 接口

系统会自动：

1. **创建数据库迁移**
   ```sql
   -- migrations/001_create_users_table.sql
   CREATE TABLE users (
     id BIGINT PRIMARY KEY AUTO_INCREMENT,
     email VARCHAR(255) UNIQUE NOT NULL,
     password_hash VARCHAR(255) NOT NULL,
     created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
   );
   ```

2. **创建 User 模型**
   ```typescript
   // src/models/user.ts
   import bcrypt from 'bcrypt';
   
   export class User {
     id?: number;
     email: string;
     password_hash: string;
     created_at?: Date;
   
     async hashPassword(password: string) {
       this.password_hash = await bcrypt.hash(password, 10);
     }
   
     async comparePassword(password: string): Promise<boolean> {
       return bcrypt.compare(password, this.password_hash);
     }
   }
   ```

3. **实现注册 API**
   ```typescript
   // src/api/auth/register.ts
   import { Router } from 'express';
   import jwt from 'jsonwebtoken';
   import { User } from '../../models/user';
   
   const router = Router();
   
   router.post('/api/auth/register', async (req, res) => {
     try {
       const { email, password } = req.body;
       
       // 验证输入
       if (!email || !password) {
         return res.status(400).json({ error: '邮箱和密码不能为空' });
       }
       
       // 检查邮箱是否已存在
       const existing = await User.findByEmail(email);
       if (existing) {
         return res.status(400).json({ error: '邮箱已存在' });
       }
       
       // 创建用户
       const user = new User();
       user.email = email;
       await user.hashPassword(password);
       await user.save();
       
       // 生成 JWT token
       const token = jwt.sign(
         { id: user.id, email: user.email },
         process.env.JWT_SECRET,
         { expiresIn: '7d' }
       );
       
       res.json({ token, user: { id: user.id, email: user.email } });
     } catch (error) {
       res.status(500).json({ error: '注册失败' });
     }
   });
   
   export default router;
   ```

### 3.2 实现注册表单

系统会自动创建：

```vue
<!-- src/pages/register.vue -->
<template>
  <div class="register-page">
    <el-card class="register-card">
      <h2>用户注册</h2>
      <el-form :model="form" :rules="rules" ref="formRef">
        <el-form-item label="邮箱" prop="email">
          <el-input 
            v-model="form.email" 
            type="email"
            placeholder="请输入邮箱"
          />
        </el-form-item>
        
        <el-form-item label="密码" prop="password">
          <el-input 
            v-model="form.password" 
            type="password"
            placeholder="请输入密码"
            show-password
          />
        </el-form-item>
        
        <el-form-item>
          <el-button 
            type="primary" 
            @click="handleRegister"
            :loading="loading"
            style="width: 100%"
          >
            注册
          </el-button>
        </el-form-item>
      </el-form>
    </el-card>
  </div>
</template>

<script setup lang="ts">
import { ref, reactive } from 'vue';
import { useRouter } from 'vue-router';
import { ElMessage } from 'element-plus';
import { registerApi } from '@/api/auth';

const router = useRouter();
const formRef = ref();
const loading = ref(false);

const form = reactive({
  email: '',
  password: ''
});

const rules = {
  email: [
    { required: true, message: '请输入邮箱', trigger: 'blur' },
    { type: 'email', message: '邮箱格式不正确', trigger: 'blur' }
  ],
  password: [
    { required: true, message: '请输入密码', trigger: 'blur' },
    { min: 6, message: '密码至少6位', trigger: 'blur' }
  ]
};

const handleRegister = async () => {
  try {
    await formRef.value.validate();
    loading.value = true;
    
    const { token, user } = await registerApi(form.email, form.password);
    
    // 保存 token
    localStorage.setItem('token', token);
    localStorage.setItem('user', JSON.stringify(user));
    
    ElMessage.success('注册成功！');
    router.push('/');
  } catch (error: any) {
    ElMessage.error(error.message || '注册失败');
  } finally {
    loading.value = false;
  }
};
</script>

}
</style>
```

## 关键收获

### 1. 智能路由的威力

你没有手动调用任何 Skill，只是用自然语言描述需求，系统就：
- ✅ 自动选择正确的 Skill 链
- ✅ 自动推进流程状态机
- ✅ 自动验证前置条件

### 2. TODO 驱动的好处

TODO 不只是任务清单，它包含：
- 写集边界（防止并行冲突）
- 验收条件（自动验证完成度）
- 依赖关系（自动检查可领取性）
- 交付面（确保不遗漏）

### 3. 一次 Review 原则

每个功能只做一次完整 Review，后续 Skill 复用结果：
- 发现问题 → 自动修复 → 重新 Review
- Review 通过 → 后续 Skill 消费 findings
- 不重复审查同一 diff
